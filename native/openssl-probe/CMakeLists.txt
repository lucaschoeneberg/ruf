cmake_minimum_required(VERSION 3.22)
project(sip_openssl_probe C)
set(OPENSSL_USE_STATIC_LIBS TRUE)
# A changed OPENSSL_ROOT_DIR must replace paths from a previous configure.
foreach(variable OPENSSL_INCLUDE_DIR OPENSSL_SSL_LIBRARY OPENSSL_CRYPTO_LIBRARY
    LIB_EAY LIB_EAY_DEBUG LIB_EAY_RELEASE SSL_EAY SSL_EAY_DEBUG SSL_EAY_RELEASE)
  unset(${variable} CACHE)
endforeach()
find_package(OpenSSL 3 REQUIRED)
if(APPLE)
  if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    message(FATAL_ERROR "Set CMAKE_OSX_DEPLOYMENT_TARGET before checking OpenSSL")
  endif()
  foreach(library "${OPENSSL_SSL_LIBRARY}" "${OPENSSL_CRYPTO_LIBRARY}")
    execute_process(COMMAND otool -l "${library}"
      OUTPUT_VARIABLE commands ERROR_VARIABLE inspection_error RESULT_VARIABLE status)
    if(NOT status EQUAL 0)
      message(FATAL_ERROR "Cannot inspect OpenSSL archive ${library}: ${inspection_error}")
    endif()
    # Inspect every archive member, including older LC_VERSION_MIN_MACOSX objects.
    string(REGEX MATCHALL "minos [0-9]+\\.[0-9]+(\\.[0-9]+)?" minimums "${commands}")
    string(REGEX MATCHALL "cmd LC_VERSION_MIN_MACOSX[\r\n\t ]+cmdsize [0-9]+[\r\n\t ]+version [0-9]+\\.[0-9]+(\\.[0-9]+)?" legacy_minimums "${commands}")
    foreach(minimum IN LISTS legacy_minimums)
      string(REGEX REPLACE ".*version " "version " minimum "${minimum}")
      list(APPEND minimums "${minimum}")
    endforeach()
    if(NOT minimums)
      message(FATAL_ERROR "No macOS deployment metadata in OpenSSL archive ${library}")
    endif()
    foreach(minimum IN LISTS minimums)
      string(REGEX REPLACE "^[^ ]+ " "" version "${minimum}")
      if(version VERSION_GREATER CMAKE_OSX_DEPLOYMENT_TARGET)
        message(FATAL_ERROR "OpenSSL ${library} requires macOS ${version}; Ruf targets ${CMAKE_OSX_DEPLOYMENT_TARGET}. Build compatible static OpenSSL and set OPENSSL_ROOT_DIR; see native/README.md.")
      endif()
    endforeach()
  endforeach()
endif()
# Imported target paths resolve MSVC's configuration-specific library names too.
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/openssl-$<CONFIG>.json"
  CONTENT "{\"ssl\":\"$<TARGET_FILE:OpenSSL::SSL>\",\"crypto\":\"$<TARGET_FILE:OpenSSL::Crypto>\",\"include\":\"${OPENSSL_INCLUDE_DIR}\",\"version\":\"${OPENSSL_VERSION}\"}\n")
